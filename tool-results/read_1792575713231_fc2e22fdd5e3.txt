    56→interface AdvancedParams {

    58→  detectionFloor: number;
    59→  significanceK: number;

    61→  detrend: boolean;

    63→
    64→const DEFAULT_ADVANCED: AdvancedParams = {
    65→  coherenceThreshold: 0.5,
    66→  detectionFloor: 0.01,
    67→  significanceK: 3,
    68→  referenceStrategy: 'farField',
    69→  detrend: false,
    70→};
    71→
    72→/**
    73→ * Display / default-selection rank: a real, ready, measurable dataset first,
    74→ * then the real evidence view, then demonstrations, then unavailable data.
    75→ * Stable, so the registry order is preserved within each class.
    76→ */
    77→function readinessRank(d: DatasetListEntry): number {
    78→  if (d.availability.state !== 'available') return 3;
    79→  if (d.dataClass === 'synthetic') return 2;
    80→  if (d.mode === 'evidence') return 1;
    81→  return 0;
    82→}
    83→
    84→/**
    85→ * Real processing stages, mirroring src/lib/science/pipeline.ts (Stages A–E).
    86→ * The server runs them in a single request, so while a run is in flight the
    87→ * list is shown as "in progress" as a whole — no per-stage completion is
    88→ * faked. Checkmarks appear only after the result has actually returned.
    89→ */
    90→function processingStages(mode: 'quantitative' | 'evidence'): Array<{ label: string; detail: string }> {
    91→  if (mode === 'evidence') {
    92→    return [
    93→      { label: 'Reading the real catalog records', detail: 'Loading the cached NASA and USGS records for the event and the radar data that covers it.' },
    94→      { label: 'Checking the latest data availability', detail: 'Asking the USGS earthquake catalog for the latest picture of the event sequence.' },
    95→      { label: 'Computing the earthquake model prediction', detail: 'Calculating the ground movement a standard earthquake model expects, from the published fault information.' },
    96→    ];
    97→  }
    98→  return [
    99→    { label: 'Checking the radar data', detail: 'Reading the measurement files and setting aside pixels whose signal is too unreliable to measure.' },
   100→    { label: 'Measuring changes in the radar signal', detail: 'Comparing the two radar images and converting the difference into apparent movement toward or away from the satellite.' },
   101→    { label: 'Estimating measurement uncertainty', detail: 'Estimating how much of the apparent change could come from measurement noise and uncorrected atmospheric effects.' },
   102→    { label: 'Checking whether the pattern is a possible real change', detail: 'Flagging areas where the apparent change clearly exceeds the uncertainty, and testing the rule against simulated no-change scenes.' },
   103→  ];
   104→}
   105→
   106→function layerLabel(l: LayerKind, isPrediction: boolean): string {
   107→  if (l === 'displacement') return isPrediction ? 'Model prediction' : 'Measured movement';
   108→  if (l === 'coherence') return 'Signal quality';
   109→  if (l === 'sigma') return 'Uncertainty';
   110→  return 'Measured areas';
   111→}
   112→
   113→function layerInterpretation(l: LayerKind, isPrediction: boolean): string {
   114→  if (l === 'displacement') {
   115→    return isPrediction
   116→      ? 'Colors show where a standard earthquake model predicts the ground should have moved. This is a calculation, not a measurement of this event.'
   117→      : 'Colors show how far each patch of ground appears to have moved toward or away from the satellite between the two radar images. Areas without color could not be measured reliably.';
   118→  }
   119→  if (l === 'coherence') return 'Darker areas have a stronger, more reliable radar signal; lighter areas are noisier. Very light areas were excluded from the analysis.';
   120→  if (l === 'sigma') return 'Darker areas have more uncertain measurements. Changes smaller than this uncertainty cannot be distinguished from noise.';
   121→  return 'Gray shows where a valid measurement exists. Transparent areas were excluded (no reliable signal).';
   122→}
